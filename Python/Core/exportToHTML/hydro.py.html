import pickle

from pylab import *

import nemoh
import tool_box as tb


class hydro_dynamic_properties(object):
    def __init__(self, dof, w, dir, fio, sym):
        self._sym = sym
        self._idof = np.array([i for i, x in enumerate(dof) if x])
        sys.stdout.write("\nInit hydro:\n")
        # -------------------------------------------------------
        sys.stdout.write("\t(1)Get forces\n")
        self._fe = nemoh.get_fe(fio.nemoh_results, dof, w, dir)
        # -------------------------------------------------------
        sys.stdout.write("\t(2)Get static mass and waterplane stiffness\n")
        self._m6x6, self._k6x6 = tb.load_M_and_K(fio.data_io_dir)
        self._m = self._m6x6[self._idof[:, None], self._idof[None, :]]
        self._k = self._k6x6[self._idof[:, None], self._idof[None, :]]
        # -------------------------------------------------------
        sys.stdout.write("\t(3)Get added mass and damping\n")
        self._ma, self._c_hyd = nemoh.get_ab(fio.nemoh_results, dof, w, dir)
        # -------------------------------------------------------
        nproblems = (len(dir) + sum(dof)) * len(w)
        with open(fio.nemoh_root.joinpath('Normalvelocities.dat')) as f:
            if not int(f.readline()) == nproblems:
                print('nproblems mismatch')
                exit()
        # -------------------------------------------------------
        print('\n')
        f_sec_file = fio.data_io_dir.joinpath('f_sec.pkl')
        if f_sec_file.is_file() and 1:
            self._fsec = pickle.load(open(f_sec_file, 'rb'))
            sys.stdout.write("\t(4)Get pressures\n")

        else:
            self._fsec = np.zeros((nproblems, 6), dtype=np.complex)
            for i in range(nproblems):
                c = int(((i + 1) * 100) / nproblems)
                sys.stdout.write("\r\t(4)Processing pressures [ %d" % c + "% ]")
                sys.stdout.flush()

                self._fsec[i, :] = nemoh.get_f_yz(fio.nemoh_results, i + 1, self._sym)

            pickle.dump(self._fsec, open(f_sec_file, "wb"))
            sys.stdout.write("\n")
        # -------------------------------------------------------

        self._w = w
        self._dof = dof
        self._dir = dir

        print('\nNproblems = {}'.format(nproblems))

    @property
    def w(self):
        return self._w

    def sdof_val(self, m, dir_index, dof_index):
        return m[dir_index, :, dof_index]

    def calcRAO(self, fe, m, ma, c, k, w):
        rao = np.absolute(fe / (-w ** 2 * (m + ma) + 1j * w * (c) + k))
        return rao

        # rao = np.zeros(w.shape[0], dtype='complex')

        # return rao


    def getRAO(self, sel_dof, sel_dir):
        idir = self.get_dir_index(sel_dir)
        idof = self.get_dof_index(sel_dof)
        fe = self._fe[idir, :, idof]
        m = self._m[idof][idof]
        ma = self._ma[idir, :,idof, idof]
        c = self._c_hyd[idir, :,idof, idof]
        k = self._k[idof][idof]
        return self.calcRAO(fe, m, ma, c, k, self._w)

    @property
    def available_dofs(self):
        return [i + 1 for i, e in enumerate(self._dof) if e != 0]

    @property
    def available_dirs(self):
        return [e for i, e in enumerate(self._dir)]

    def get_dof_index(self, sel_dof):
        dof_index = 0
        if self._dof[sel_dof - 1] == 1:
            return (sum(self._dof[:sel_dof]) - 1)

def spec_wave(hs, wp, w, gamma=None):
    sig_a = 0.07
    sig_b = 0.09
    delta_sig = sig_b - sig_a

    tp = 2 * np.pi / wp
    x = tp / np.sqrt(hs)
    if gamma == None:
        if x <= 3.6:
            gamma = 5
        elif x < 5:
            gamma = np.exp(5.75 - 1.15 * x)
        else:
            gamma = 1
        # print('Gamma {}'.format(gamma))

    a_gamma = 1 - 0.287 * np.log(gamma)

    def spec_pm(w):
        return (5 / 16) * (hs ** 2) * (wp ** 4) * (w ** (-5)) * np.exp(-(5 / 4) * ((w / wp) ** (-4)))

    def spec_jonswap(w):
        def sig(w):
            return sig_a if w <= wp else sig_b

        sig_ab = np.array(list(map(sig, w)))

        return a_gamma * spec_pm(w) * gamma ** np.exp(-0.5 * ((w - wp) / sig_ab * wp))

    if not gamma == 1:
        spec = spec_jonswap
    else:
        spec = spec_pm

    return spec(w)


def spec_response(h, w, hs, wp):
    s = spec_wave(hs, wp, w)
    return h ** 2 * s
